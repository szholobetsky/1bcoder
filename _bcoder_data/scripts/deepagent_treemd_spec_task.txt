# this one, unlike deepagent_md.
#
# Required params:
#   plan = the existing plan name (e.g. plan1)
#
# Usage:
#   1bcoder --host openai://localhost:1234 --model qwen3-4b-instruct-2507 ^
#       --scriptapply deepagent_spec_task.txt --param plan=plan1

/param think false
